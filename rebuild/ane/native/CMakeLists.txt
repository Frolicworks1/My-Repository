cmake_minimum_required(VERSION 3.22)
project(ttpixel_core LANGUAGES C CXX)

add_library(ttpixel_core SHARED ttpixel_core.cpp)
target_include_directories(ttpixel_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_features(ttpixel_core PRIVATE cxx_std_17)
target_compile_options(ttpixel_core PRIVATE -Wall -Wextra -Werror)

include(CTest)
if(BUILD_TESTING)
    add_executable(test_ttpixel_core test_ttpixel_core.cpp ttpixel_core.cpp)
    target_include_directories(test_ttpixel_core PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
    target_compile_features(test_ttpixel_core PRIVATE cxx_std_17)
    target_compile_options(test_ttpixel_core PRIVATE -Wall -Wextra -Werror)
    add_test(NAME ttpixel_core_contract COMMAND test_ttpixel_core)
endif()
